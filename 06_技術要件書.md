### 2.4 tRPC設定

#### server/api/trpc.ts

```typescript
import { initTRPC, TRPCError } from '@trpc/server';
import { type CreateNextContextOptions } from '@trpc/server/adapters/next';
import { getServerSession } from 'next-auth';
import { prisma } from '@/server/db';
import { authOptions } from '@/app/api/auth/[...nextauth]/route';

export const createTRPCContext = async (opts: CreateNextContextOptions) => {
  const session = await getServerSession(authOptions);

  return {
    session,
    db: prisma,
  };
};

const t = initTRPC.context<typeof createTRPCContext>().create();

export const createTRPCRouter = t.router;
export const publicProcedure = t.procedure;

// 認証が必要なプロシージャ
export const protectedProcedure = t.procedure.use(
  async ({ ctx, next }) => {
    if (!ctx.session?.user) {
      throw new TRPCError({ code: 'UNAUTHORIZED' });
    }

    return next({
      ctx: {
        session: ctx.session,
        userId: ctx.session.user.id,
        db: ctx.db,
      },
    });
  }
);
```

# src/main.py
from fastapi import FastAPI, HTTPException
from fastapi.middleware.cors import CORSMiddleware
from pydantic import BaseModel
from loguru import logger

app = FastAPI(
    title="Financial Intelligence Engine",
    version="1.0.0",
    docs_url="/api/docs",
)

# CORS設定
app.add_middleware(
    CORSMiddleware,
    allow_origins=[
        "https://your-frontend.run.app",
        "http://localhost:3000",  # 開発環境
    ],
    allow_credentials=True,
    allow_methods=["*"],
    allow_headers=["*"],
)

# ヘルスチェック
@app.get("/health")
async def health_check():
    return {"status": "healthy"}

# データ収集エンドポイント
@app.post("/api/collect/twitter")
async def collect_twitter():
    """
    Cloud Schedulerから呼ばれるTwitter監視エンドポイント
    """
    try:
        from src.collectors.twitter_collector import TwitterCollector
        collector = TwitterCollector()
        result = await collector.collect()
        return {"status": "success", "collected": result}
    except Exception as e:
        logger.error(f"Twitter collection failed: {e}")
        raise HTTPException(status_code=500, detail=str(e))

@app.post("/api/collect/marketcap")
async def collect_marketcap():
    """
    時価総額データ収集
    """
    try:
        from src.collectors.fmp_collector import FMPCollector
        collector = FMPCollector()
        result = await collector.collect_ranking()
        return {"status": "success", "data": result}
    except Exception as e:
        logger.error(f"Market cap collection failed: {e}")
        raise HTTPException(status_code=500, detail=str(e))

if __name__ == "__main__":
    import uvicorn
    uvicorn.run(app, host="0.0.0.0", port=8000)

### 8.1 必須ソフトウェア

```bash
# Node.js
brew install node@20

# Python
brew install python@3.11

# PostgreSQL (ローカル開発)
brew install postgresql@15

# Docker
brew install --cask docker

# gcloud CLI
brew install --cask google-cloud-sdk
```

---

### 8.2 プロジェクト初期化

```bash
# 1. リポジトリクローン
git clone https://github.com/your-org/fin-intel.git
cd fin-intel

# 2. フロントエンド
cd frontend
npm install
npx prisma generate
cp .env.example .env.local
# .env.localを編集

# 3. バックエンド
cd ../backend
python -m venv .venv
source .venv/bin/activate
pip install -r requirements.txt
cp .env.example .env
# .envを編集

# 4. データベース起動
docker-compose up -d postgres

# 5. マイグレーション
cd ../frontend
npx prisma migrate dev

# 6. 開発サーバー起動
npm run dev  # フロントエンド（http://localhost:3000）

cd ../backend
uvicorn src.main:app --reload  # バックエンド（http://localhost:8000）
```
