import tweepy
from datetime import datetime, timedelta

class TwitterCollector:
    def __init__(self, bearer_token: str):
        self.client = tweepy.Client(bearer_token=bearer_token)

    async def collect_tweets(self, accounts: list[str]) -> list[Tweet]:
        """
        指定アカウントの最新ツイートを収集
        """
        tweets = []
        for account in accounts:
            user = self.client.get_user(username=account)
            user_tweets = self.client.get_users_tweets(
                id=user.data.id,
                max_results=5,
                tweet_fields=['created_at', 'public_metrics']
            )
            tweets.extend(user_tweets.data)
        return tweets
```

# backend/src/ai/translator.py
from anthropic import Anthropic

class FinancialTranslator:
    def __init__(self, api_key: str):
        self.client = Anthropic(api_key=api_key)

    async def translate_and_summarize(
        self,
        original_text: str
    ) -> dict:
        """
        翻訳・要約・センチメント分析
        """
        prompt = self._build_prompt(original_text)

        response = self.client.messages.create(
            model="claude-3-5-sonnet-20241022",
            max_tokens=1024,
            messages=[{
                "role": "user",
                "content": prompt
            }]
        )

        result = self._parse_response(response.content[0].text)
        return result
```

##### Day 18-19: LangGraph実装
```python
# backend/src/ai/workflow.py
from langgraph.graph import StateGraph, END

class AnalysisWorkflow:
    def __init__(self):
        self.workflow = self._build_workflow()

    def _build_workflow(self) -> StateGraph:
        workflow = StateGraph(AnalysisState)

        # ノード追加
        workflow.add_node("translate", self.translate_node)
        workflow.add_node("summarize", self.summarize_node)
        workflow.add_node("sentiment", self.sentiment_node)
        workflow.add_node("quality_check", self.quality_check_node)

        # フロー定義
        workflow.add_edge("translate", "summarize")
        workflow.add_edge("summarize", "sentiment")
        workflow.add_edge("sentiment", "quality_check")

        # 条件分岐
        workflow.add_conditional_edges(
            "quality_check",
            self._should_retry,
            {
                "retry": "translate",
                "end": END
            }
        )

        return workflow.compile()
```

from llama_index import VectorStoreIndex, Document

class FinancialTermRAG:

        self.index = self._build_index()

    def _build_index(self):

// app/api/auth/[...nextauth]/route.ts
import NextAuth from "next-auth";
import CredentialsProvider from "next-auth/providers/credentials";
import { compare } from "bcryptjs";

export const authOptions = {
  providers: [
    CredentialsProvider({
      name: "Credentials",
      credentials: {
        email: { label: "Email", type: "email" },
        password: { label: "Password", type: "password" }
      },
      async authorize(credentials) {
        const user = await prisma.user.findUnique({
          where: { email: credentials.email }
        });

        if (!user) return null;

        const isValid = await compare(
          credentials.password,
          user.passwordHash
        );

        if (!isValid) return null;

        return { id: user.id, email: user.email, name: user.name };
      }
    })
  ],
  pages: {
    signIn: '/auth/signin',
  },
  callbacks: {
    async jwt({ token, user }) {
      if (user) {
        token.id = user.id;
      }
      return token;
    },
    async session({ session, token }) {
      session.user.id = token.id;
      return session;
    }
  }
};

const handler = NextAuth(authOptions);
export { handler as GET, handler as POST };
```

// server/api/routers/news.ts
export const newsRouter = createTRPCRouter({
  getAll: protectedProcedure
    .input(
      z.object({
        sector: z.string().optional(),
        sentiment: z.enum(['positive', 'negative', 'neutral']).optional(),
        limit: z.number().default(20),
        offset: z.number().default(0),
      })
    )
    .query(async ({ ctx, input }) => {
      const news = await ctx.db.news.findMany({
        where: {
          sector: input.sector,
          sentiment: input.sentiment,
        },
        take: input.limit,
        skip: input.offset,
        orderBy: { publishedAt: 'desc' },
      });

      const total = await ctx.db.news.count();

      return {
        news,
        total,
        hasMore: input.offset + input.limit < total,
      };
    }),
});
```
